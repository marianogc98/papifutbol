import React from 'react';
import { Badge } from '@/components/ui/badge';

interface Equipo {
  id: number;
  nombre: string;
  escudo?: string;
  slug?: string;
}

interface Goleador {
  equipo: 'local' | 'visitante';
  nombre: string;
  apellido: string;
  minuto: number;
}

interface Partido {
  id: number;
  fechaHora?: string;
  equipoLocal: Equipo;
  equipoVisitante: Equipo;
  golesLocal: number;
  golesVisitante: number;
  estado: 'programado' | 'jugando' | 'finalizado';
  goleadores: Goleador[];
}

interface TablaResultadosProps {
  partidos: Partido[];
  titulo?: string;
  mostrarVerTodos?: boolean;
  onVerTodos?: () => void;
}

const TablaResultados: React.FC<TablaResultadosProps> = ({ 
  partidos, 
  titulo = "Últimos Resultados",
  mostrarVerTodos = false,
  onVerTodos
}) => {
  const formatDateTime = (dateString: string) => {
    const date = new Date(dateString);
    return date.toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit' });
  };

  const getGoleadores = (partido: Partido, equipo: 'local' | 'visitante') => {
    return partido.goleadores.filter(g => g.equipo === equipo);
  };

  const formatGoleadores = (goleadores: Goleador[]) => {
    const golesPorJugador = goleadores.reduce((acc, gol) => {
      const key = `${gol.nombre} ${gol.apellido}`;
      if (!acc[key]) {
        acc[key] = { nombre: gol.nombre, apellido: gol.apellido, cantidad: 0 };
      }
      acc[key].cantidad++;
      return acc;
    }, {} as Record<string, { nombre: string; apellido: string; cantidad: number }>);

    return Object.values(golesPorJugador).map((jugador, idx, arr) => (
      <span key={idx}>
        {jugador.nombre[0]}. {jugador.apellido}
        {jugador.cantidad > 1 && ` (${jugador.cantidad})`}
        {idx < arr.length - 1 && ', '}
      </span>
    ));
  };

  return (
    <div className="w-full">
      {/* Header */}
      {titulo && (
        <div className="flex items-center justify-between mb-6">
          <h2 className="text-3xl font-bold text-slate-800">{titulo}</h2>
          {mostrarVerTodos && (
            <button 
              onClick={onVerTodos}
              className="flex items-center gap-2 text-[#852024] hover:text-[#CA3F42] font-medium transition-colors"
            >
              Ver todos
              <svg className="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M9 5l7 7-7 7" />
              </svg>
            </button>
          )}
        </div>
      )}

      {/* Vista Desktop */}
      <div className="hidden md:block bg-white border rounded-xl overflow-hidden shadow-lg">
        <table className="w-full">
          <thead className="bg-slate-50 border-b">
            <tr>
              <th className="px-6 py-4 text-left text-xs font-medium text-slate-600 uppercase tracking-wider w-20">
                Hora
              </th>
              <th className="px-6 py-4 text-left text-xs font-medium text-slate-600 uppercase tracking-wider">
                Local
              </th>
              <th className="px-6 py-4 text-center text-xs font-medium text-slate-600 uppercase tracking-wider w-24">
                Resultado
              </th>
              <th className="px-6 py-4 text-right text-xs font-medium text-slate-600 uppercase tracking-wider">
                Visitante
              </th>
              <th className="px-6 py-4 text-left text-xs font-medium text-slate-600 uppercase tracking-wider w-32">
                Estado
              </th>
            </tr>
          </thead>
          <tbody>
            {partidos.map((partido) => {
              const goleadoresLocal = getGoleadores(partido, 'local');
              const goleadoresVisitante = getGoleadores(partido, 'visitante');
              const mostrarResultado = partido.estado === 'finalizado' || partido.estado === 'jugando';

              return (
                <tr key={partido.id} className="border-b hover:bg-slate-50 transition-colors">
                  <td className="px-6 py-4">
                    {partido.fechaHora ? (
                      <span className="text-sm text-slate-600 font-medium">
                        {formatDateTime(partido.fechaHora)}
                      </span>
                    ) : (
                      <span className="text-sm text-slate-400">-</span>
                    )}
                  </td>
                  
                  <td className="px-6 py-4">
                    <div className="flex flex-col gap-1">
                      <a
                        href={`/equipo/${partido.equipoLocal?.slug || partido.equipoLocal?.id}`}
                        className="flex items-center gap-3 hover:underline w-fit group"
                      >
                        {partido.equipoLocal?.escudo && (
                          <img
                            src={partido.equipoLocal.escudo}
                            alt={partido.equipoLocal.nombre}
                            className="w-10 h-10 object-contain"
                          />
                        )}
                        <span className="font-medium text-slate-800 group-hover:text-[#852024] transition-colors">
                          {partido.equipoLocal?.nombre}
                        </span>
                      </a>
                      {goleadoresLocal.length > 0 && (
                        <div className="text-xs text-slate-500 pl-13">
                          ⚽ {formatGoleadores(goleadoresLocal)}
                        </div>
                      )}
                    </div>
                  </td>
                  
                  <td className="px-6 py-4 text-center">
                    {mostrarResultado ? (
                      <span className="font-bold text-2xl text-slate-800">
                        {partido.golesLocal} - {partido.golesVisitante}
                      </span>
                    ) : (
                      <span className="text-slate-400 text-sm">vs</span>
                    )}
                  </td>
                  
                  <td className="px-6 py-4">
                    <div className="flex flex-col gap-1 items-end">
                      <a
                        href={`/equipo/${partido.equipoVisitante?.slug || partido.equipoVisitante?.id}`}
                        className="flex items-center gap-3 justify-end hover:underline w-fit group"
                      >
                        <span className="font-medium text-slate-800 group-hover:text-[#852024] transition-colors">
                          {partido.equipoVisitante?.nombre}
                        </span>
                        {partido.equipoVisitante?.escudo && (
                          <img
                            src={partido.equipoVisitante.escudo}
                            alt={partido.equipoVisitante.nombre}
                            className="w-10 h-10 object-contain"
                          />
                        )}
                      </a>
                      {goleadoresVisitante.length > 0 && (
                        <div className="text-xs text-slate-500 pr-13">
                          {formatGoleadores(goleadoresVisitante)} ⚽
                        </div>
                      )}
                    </div>
                  </td>
                  
                  <td className="px-6 py-4">
                    {partido.estado === 'jugando' && (
                      <Badge className="bg-green-600 hover:bg-green-700">
                        <span className="w-2 h-2 bg-white rounded-full mr-2 inline-block animate-pulse"></span>
                        En Vivo
                      </Badge>
                    )}
                    {partido.estado === 'finalizado' && (
                      <Badge className="bg-slate-500 hover:bg-slate-600">
                        Finalizado
                      </Badge>
                    )}
                  </td>
                </tr>
              );
            })}
          </tbody>
        </table>
      </div>

      {/* Vista Mobile */}
      <div className="md:hidden space-y-4">
        {partidos.map((partido) => {
          const goleadoresLocal = getGoleadores(partido, 'local');
          const goleadoresVisitante = getGoleadores(partido, 'visitante');
          const mostrarResultado = partido.estado === 'finalizado' || partido.estado === 'jugando';

          return (
            <div key={partido.id} className="bg-white border rounded-xl p-4 shadow-md">
              <div className="flex justify-between items-center mb-4">
                <span className="text-sm font-medium text-slate-600">
                  {partido.fechaHora ? formatDateTime(partido.fechaHora) : '-'}
                </span>
                {partido.estado === 'jugando' && (
                  <Badge className="bg-green-600 hover:bg-green-700 text-xs">
                    <span className="w-1.5 h-1.5 bg-white rounded-full mr-1 inline-block animate-pulse"></span>
                    En Vivo
                  </Badge>
                )}
                {partido.estado === 'finalizado' && (
                  <Badge className="bg-slate-500 hover:bg-slate-600 text-xs">
                    Finalizado
                  </Badge>
                )}
              </div>

              <div className="flex items-start justify-between mb-2">
                <a
                  href={`/equipo/${partido.equipoLocal?.slug || partido.equipoLocal?.id}`}
                  className="flex items-center gap-2 hover:underline flex-1"
                >
                  {partido.equipoLocal?.escudo && (
                    <img
                      src={partido.equipoLocal.escudo}
                      alt={partido.equipoLocal.nombre}
                      className="w-9 h-9 object-contain flex-shrink-0"
                    />
                  )}
                  <span className="font-medium text-sm text-slate-800">
                    {partido.equipoLocal?.nombre}
                  </span>
                </a>
                {mostrarResultado && (
                  <span className="font-bold text-xl ml-2 text-slate-800">
                    {partido.golesLocal}
                  </span>
                )}
              </div>
              
              {goleadoresLocal.length > 0 && (
                <div className="text-xs text-slate-500 ml-11 mb-3">
                  ⚽ {formatGoleadores(goleadoresLocal)}
                </div>
              )}

              <div className="flex items-start justify-between mb-2">
                <a
                  href={`/equipo/${partido.equipoVisitante?.slug || partido.equipoVisitante?.id}`}
                  className="flex items-center gap-2 hover:underline flex-1"
                >
                  {partido.equipoVisitante?.escudo && (
                    <img
                      src={partido.equipoVisitante.escudo}
                      alt={partido.equipoVisitante.nombre}
                      className="w-9 h-9 object-contain flex-shrink-0"
                    />
                  )}
                  <span className="font-medium text-sm text-slate-800">
                    {partido.equipoVisitante?.nombre}
                  </span>
                </a>
                {mostrarResultado && (
                  <span className="font-bold text-xl ml-2 text-slate-800">
                    {partido.golesVisitante}
                  </span>
                )}
              </div>

              {goleadoresVisitante.length > 0 && (
                <div className="text-xs text-slate-500 ml-11">
                  ⚽ {formatGoleadores(goleadoresVisitante)}
                </div>
              )}
            </div>
          );
        })}
      </div>
    </div>
  );
};

export default TablaResultados;
